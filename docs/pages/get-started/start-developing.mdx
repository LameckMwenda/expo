const FirebaseContext = createContext(null);

// Firebase Provider component to initialize Firebase and manage auth state
const FirebaseProvider = ({ children }) => {
  const [db, setDb] = useState(null);
  const [auth, setAuth] = useState(null);
  const [userId, setUserId] = useState(null);
  const [loading, setLoading] = useState(true);
  const [authError, setAuthError] = useState(null);

  useEffect(() => {
    try {
      const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-biotrack-lab-app';
      const firebaseConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : {};

      if (!firebaseConfig || Object.keys(firebaseConfig).length === 0) {
        console.error("Firebase config is missing or empty.");
        setAuthError("Firebase is not configured. Please ensure __firebase_config is set.");
        setLoading(false);
        return;
      }

      const app = initializeApp(firebaseConfig);
      const firestoreDb = getFirestore(app);
      const firebaseAuth = getAuth(app);

      setDb(firestoreDb);
      setAuth(firebaseAuth);

      const initialSignIn = async () => {
        try {
          if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
            await signInWithCustomToken(firebaseAuth, __initial_auth_token);
            console.log("Signed in with custom token.");
          } else {
            await signInAnonymously(firebaseAuth);
            console.log("Signed in anonymously.");
          }
        } catch (error) {
          console.error("Firebase initial sign-in error:", error);
          setAuthError(`Authentication failed: ${error.message}`);
        } finally {
          setLoading(false);
        }
      };

      initialSignIn();

      const unsubscribe = onAuthStateChanged(firebaseAuth, (user) => {
        if (user) {
          setUserId(user.uid);
          console.log("Auth state changed, user ID:", user.uid);
        } else {
          setUserId(null);
          console.log("Auth state changed, no user.");
        }
        setLoading(false);
      });

      return () => unsubscribe();
    } catch (error) {
      console.error("Firebase initialization failed:", error);
      setAuthError(`Firebase initialization failed: ${error.message}`);
      setLoading(false);
    }
  }, []);

  const value = { db, auth, userId, loading, authError };

  return (
    <FirebaseContext.Provider value={value}>
      {children}
    </FirebaseContext.Provider>
  );
};

// Hook for Firestore Collection Data
const useFirestoreCollection = (collectionName) => {
  const { db, userId } = useFirebase();
  const [data, setData] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    if (!db || !userId) {
      setLoading(false);
      return;
    }

    setLoading(true);
    const q = query(collection(db, `artifacts/${__app_id}/users/${userId}/${collectionName}`));
    const unsubscribe = onSnapshot(q, (snapshot) => {
      const fetchedData = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
      setData(fetchedData);
      setLoading(false);
      setError(null);
    }, (err) => {
      console.error(`Error fetching ${collectionName}:`, err);
      setError(`Failed to load ${collectionName}.`);
      setLoading(false);
    });

    return () => unsubscribe();
  }, [db, userId, collectionName]);

  const addDocument = useCallback(async (docData) => {
    if (!db || !userId) return;
    try {
      await addDoc(collection(db, `artifacts/${__app_id}/users/${userId}/${collectionName}`), {
        ...docData,
        createdAt: serverTimestamp(),
        userId: userId
      });
      setError(null);
    } catch (err) {
      console.error(`Error adding document to ${collectionName}:`, err);
      setError(`Failed to add item to ${collectionName}.`);
    }
  }, [db, userId, collectionName]);

  const updateDocument = useCallback(async (id, updateData) => {
    if (!db || !userId) return;
    try {
      await updateDoc(doc(db, `artifacts/${__app_id}/users/${userId}/${collectionName}`, id), {
        ...updateData,
        lastUpdated: serverTimestamp()
      });
      setError(null);
    } catch (err) {
      console.error(`Error updating document in ${collectionName}:`, err);
      setError(`Failed to update item in ${collectionName}.`);
    }
  }, [db, userId, collectionName]);

  const deleteDocument = useCallback(async (id) => {
    if (!db || !userId) return;
    try {
      await deleteDoc(doc(db, `artifacts/${__app_id}/users/${userId}/${collectionName}`, id));
      setError(null);
    } catch (err) {
      console.error(`Error deleting document from ${collectionName}:`, err);
      setError(`Failed to delete item from ${collectionName}.`);
    }
  }, [db, userId, collectionName]);

  return { data, loading, error, addDocument, updateDocument, deleteDocument };
};

// Hook for Gemini API Calls
const useGeminiApi = () => {
  const [response, setResponse] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  const callApi = useCallback(async (prompt) => {
    setLoading(true);
    setResponse(null);
    setError(null);

    try {
      let chatHistory = [];
      chatHistory.push({ role: "user", parts: [{ text: prompt }] });
      const payload = { contents: chatHistory };
      const apiKey = ""; // Canvas will automatically provide it in runtime
      const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${apiKey}`;

      const fetchResponse = await fetch(apiUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload)
      });

      const result = await fetchResponse.json();
      if (result.candidates && result.candidates.length > 0 &&
          result.candidates[0].content && result.candidates[0].content.parts &&
          result.candidates[0].content.parts.length > 0) {
        const text = result.candidates[0].content.parts[0].text;
        setResponse(text);
      } else {
        setError("Failed to get response from AI. Please try again.");
        console.error("Gemini API response structure unexpected:", result);
      }
    } catch (llmError) {
      console.error("Error calling Gemini API:", llmError);
      setError("Error connecting to AI. Please check your network and try again.");
    } finally {
      setLoading(false);
    }
  }, []);

  return { response, loading, error, callApi };
};

const Auth = ({ onLoginSuccess }) => {
  const { auth, authError } = useFirebase();
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [isLogin, setIsLogin] = useState(true);
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    setLoading(true);

    try {
      if (isLogin) {
        await signInWithEmailAndPassword(auth, email, password);
        console.log("User logged in successfully.");
      } else {
        await createUserWithEmailAndPassword(auth, email, password);
        console.log("User registered successfully.");
      }
      onLoginSuccess();
    } catch (err) {
      console.error("Auth error:", err);
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center bg-gray-900 p-4">
      <div className="bg-gray-800 p-8 rounded-lg shadow-lg w-full max-w-md border border-gray-700">
        <h2 className="text-2xl font-bold mb-6 text-center text-white">
          {isLogin ? 'Lab Staff Login' : 'Register New Staff'}
        </h2>
        <form onSubmit={handleSubmit} className="space-y-4">
          <div>
            <label className="block text-gray-300 text-sm font-bold mb-2" htmlFor="email">
              Email:
            </label>
            <input
              type="email"
              id="email"
              className="shadow appearance-none border border-gray-600 bg-gray-700 rounded w-full py-2 px-3 text-white leading-tight focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-transparent rounded-md"
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              required
            />
          </div>
          <div>
            <label className="block text-gray-300 text-sm font-bold mb-2" htmlFor="password">
              Password:
            </label>
            <input
              type="password"
              id="password"

              onChange={(e) => setPassword(e.target.value)}

            />
          </div>
          {error && <p className="text-red-400 text-sm italic">{error}</p>}
          {authError && <p className="text-red-400 text-sm italic">{authError}</p>}
          <button
            type="submit"

        >
          {isLogin ? 'Need to register? Create an account' : 'Already have an account? Login'}
        </button>
      </div>
    </div>
  );
};

    }));

      setError("Please fill required fields (Item Name, Category, Quantity).");
      return;

    setError(null);

      expirationDate: '', minQuantity: 0, notes: ''
    });
    setShowAddItemModal(false);
  };

    const newQuantity = currentQuantity + change;
    if (newQuantity < 0) return;
    await updateDocument(id, { quantity: newQuantity });
  };

    await deleteDocument(id);
  };

    const lowStockItems = inventoryItems.filter(item => item.quantity <= (item.minQuantity || 0) && item.quantity > 0);
    const outOfStockItems = inventoryItems.filter(item => item.quantity === 0);

    if (lowStockItems.length === 0 && outOfStockItems.length === 0) {
      callLlmApi("No items are currently low or out of stock."); // Use callLlmApi to set response
      return;
    }

    let prompt = "Suggest reordering strategies and potential suppliers for the following lab inventory items:\n\n";
    if (outOfStockItems.length > 0) {
      prompt += "Out of Stock:\n";
      outOfStockItems.forEach(item => {
        prompt += `- ${item.itemName} (Category: ${item.category || 'N/A'})\n`;
      });
    }
    if (lowStockItems.length > 0) {
      prompt += "\nLow Stock:\n";
      lowStockItems.forEach(item => {
        prompt += `- ${item.itemName} (Current Quantity: ${item.quantity} ${item.unit || ''}, Min Quantity: ${item.minQuantity || 'N/A'}, Category: ${item.category || 'N/A'})\n`;
      });

    prompt += "\nConsider general lab suppliers and common alternatives. Provide a concise, actionable summary.";

    await callLlmApi(prompt);

  if (error) return <div className="text-center p-4 text-red-400">{error}</div>;

      <h3 className="text-xl font-semibold mb-4 text-white">Inventory Management</h3>
      <div className="mb-4 flex justify-between items-center">

          onClick={handleGetReorderSuggestions}
          className="bg-purple-600 hover:bg-purple-700 text-white py-2 px-4 rounded-md transition duration-200 flex items-center space-x-2"

            <svg className="animate-spin h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
              <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
              <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>

            '✨'
          )}

          onClick={() => setShowAddItemModal(true)}

          + New Item
        </button>
      </div>

      {llmSuggestion && (

          <h4 className="text-lg font-semibold text-white mb-2">AI Reorder Suggestions:</h4>
          <p className="text-gray-300 whitespace-pre-wrap">{llmSuggestion}</p>
        </div>
      )}
      {llmError && <p className="text-red-400 text-sm italic mb-4">{llmError}</p>}

      {inventoryItems.length === 0 ? (
        <p className="text-gray-400">No inventory items yet.</p>
      ) : (
        <div className="overflow-x-auto">
          <table className="min-w-full bg-gray-900 border border-gray-700 rounded-md text-white">
            <thead className="bg-gray-700">
              <tr>
                <th className="py-2 px-4 border-b border-gray-600 text-left text-sm font-semibold text-gray-300">ID</th>
                <th className="py-2 px-4 border-b border-gray-600 text-left text-sm font-semibold text-gray-300">Name</th>
                <th className="py-2 px-4 border-b border-gray-600 text-left text-sm font-semibold text-gray-300">Category</th>
                <th className="py-2 px-4 border-b border-gray-600 text-left text-sm font-semibold text-gray-300">Location</th>
                <th className="py-2 px-4 border-b border-gray-600 text-left text-sm font-semibold text-gray-300">Quantity</th>
                <th className="py-2 px-4 border-b border-gray-600 text-left text-sm font-semibold text-gray-300">Actions</th>
              </tr>
            </thead>

                    <div className="flex space-x-2">
                      <button
                        onClick={() => handleUpdateQuantity(item.id, item.quantity, 1)}
                        className="bg-blue-600 hover:bg-blue-700 text-white text-xs py-1 px-2 rounded-md transition duration-200"

                        +
                      </button>
                      <button
                        onClick={() => handleUpdateQuantity(item.id, item.quantity, -1)}

                        -

                        Delete

      {/* Add Item Modal */}

            <label className="block text-sm font-medium text-gray-300 mb-1">Item Name *</label>

                      className="w-full p-2 border border-gray-600 bg-gray-700 rounded-md focus:ring-blue-500 focus:border-blue-500"></textarea>

            </button>

    description: '',
    responsibleStaffName: '',
    dueDate: '',
  });

  const [currentExperimentNotes, setCurrentExperimentNotes] = useState([]);
  const [currentExperimentId, setCurrentExperimentId] = useState(null);

    const { name, value } = e.target;
    setNewExperiment(prev => ({ ...prev, [name]: value }));

    if (!newExperiment.title || !newExperiment.project || !newExperiment.startDate) {
      setError("Please fill required fields (Title, Project, Start Date).");
      return;
    }
    setError(null);
    await addDocument({
      ...newExperiment,
      status: 'Planning',
      isCompleted: false,
      isFailed: false,
      completionDate: null,
      notes: [],
    });
    setNewExperiment({ title: '', project: '', startDate: format(new Date(), 'yyyy-MM-dd'), description: '', responsibleStaffName: '', dueDate: '' });
  };

  const handleUpdateExperimentStatus = async (id, currentStatus) => {
    let newStatus = currentStatus;
    let isCompleted = false;
    let isFailed = false;
    let completionDate = null;

    if (currentStatus === 'Planning') {
      newStatus = 'InProgress';
    } else if (currentStatus === 'InProgress') {
      newStatus = 'Completed';
      isCompleted = true;
      completionDate = serverTimestamp();
    } else if (currentStatus === 'Completed') {
      newStatus = 'Failed';
      isFailed = true;
      completionDate = serverTimestamp();
    } else if (currentStatus === 'Failed') {
      newStatus = 'Planning';
      isFailed = false;
    }
    await updateDocument(id, { status: newStatus, isCompleted: isCompleted, isFailed: isFailed, completionDate: completionDate });
  };

  const handleDeleteExperiment = async (id) => {
    await deleteDocument(id);
  };

  const openNotesModal = (experiment) => {
    setCurrentExperimentNotes(experiment.notes || []);
    setCurrentExperimentId(experiment.id);
    setNewNoteText('');
    setShowNotesModal(true);
    // Clear previous LLM suggestions when opening modal for a new experiment
    if (troubleshootSuggestion) callTroubleshootApi(null); // Clear LLM response
  };

    if (!currentExperimentId || !newNoteText.trim()) return;

  }, [protocolSuggestion]);

  const handleGetTroubleshootingSuggestion = async () => {

      setError("Please describe the issue in the note field to get troubleshooting suggestions.");

  };

  if (loading) return <div className="text-center p-4 text-gray-300">Loading experiments...</div>;
  if (error) return <div className="text-center p-4 text-red-400">{error}</div>;

  return (

      <h3 className="text-xl font-semibold mb-4 text-white">Experiments Registry & Tracking</h3>
      <div className="mb-6 p-4 border border-blue-700 bg-blue-900 rounded-md text-white">
        <h4 className="text-lg font-semibold mb-3">Add New Experiment</h4>
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-4">
          <input
            type="text"
            name="title"
            placeholder="Experiment Title"
            value={newExperiment.title}
            onChange={handleNewExperimentChange}
            className="p-2 border border-gray-600 bg-gray-700 text-white rounded-md focus:ring-blue-500 focus:border-blue-500"
          />

            className="p-2 border border-gray-600 bg-gray-700 text-white rounded-md focus:ring-blue-500 focus:border-blue-500"
          />

            className="p-2 border border-gray-600 bg-gray-700 text-white rounded-md focus:ring-blue-500 focus:border-blue-500"
          />
          <div>
            <label htmlFor="startDate" className="block text-sm font-medium text-gray-300 mb-1">Start Date:</label>
            <input

              name="startDate"

              className="p-2 border border-gray-600 bg-gray-700 text-white rounded-md focus:ring-blue-500 focus:border-blue-500 w-full"

        </head>
        <body>
            <h1>BioTrack Lab System Management Data Export - ${format(new Date(), 'MMM dd,yyyy')}</h1>
      `;

      // Samples Section
      htmlContent += `<h2>Sample Tracking</h2>`;
      if (samplesData.length > 0) {
        htmlContent += `<table>
          <thead>
            <tr>
              <th>Name</th><th>Type</th><th>Status</th><th>Added On</th>
            </tr>
          </thead>
          <tbody>
            ${samplesData.map(s => `
              <tr>
                <td>${s.name}</td>
                <td>${s.type}</td>
                <td>${s.status}</td>
                <td>${s.createdAt ? format(s.createdAt.toDate(), 'MMM dd,yyyy HH:mm') : 'N/A'}</td>
              </tr>
            `).join('')}
          </tbody>
        </table>`;
      } else {
        htmlContent += `<p>No samples data available.</p>`;
      }

      // Inventory Section
      htmlContent += `<h2>Inventory Management</h2>`;
      if (inventoryData.length > 0) {
        htmlContent += `<table>
          <thead>
            <tr>
              <th>Name</th>
              <th>Category</th>
              <th>Location</th>
              <th>Quantity</th>
              <th>Unit</th>
              <th>Min. Qty</th>
              <th>Received On</th>
              <th>Expiry Date</th>
            </tr>
          </thead>
          <tbody>
            ${inventoryData.map(i => `
              <tr>
                <td>${i.itemName}</td>
                <td>${i.category}</td>
                <td>${i.storageLocation}</td>
                <td>${i.quantity}</td>
                <td>${i.unit}</td>
                <td>${i.minQuantity}</td>
                <td>${i.dateReceived ? format(parseISO(i.dateReceived), 'MMM dd,yyyy') : 'N/A'}</td>
                <td>${i.expirationDate ? format(parseISO(i.expirationDate), 'MMM dd,yyyy') : 'N/A'}</td>
              </tr>
            `).join('')}
          </tbody>
        </table>`;
      } else {
        htmlContent += `<p>No inventory data available.</p>`;
      }

      // Experiments Section
      htmlContent += `<h2>Experiments Registry</h2>`;
      if (experimentsData.length > 0) {
        htmlContent += `<table>
          <thead>
            <tr>
              <th>Title</th>
              <th>Project</th>
              <th>Responsible</th>
              <th>Start Date</th>
              <th>Due Date</th>
              <th>Status</th>
              <th>Completed On</th>
              <th>Notes Count</th>
            </tr>
          </thead>
          <tbody>
            ${experimentsData.map(e => `
              <tr>
                <td>${e.title}</td>
                <td>${e.project}</td>
                <td>${e.responsibleStaffName}</td>
                <td>${e.startDate ? format(parseISO(e.startDate), 'MMM dd,yyyy') : 'N/A'}</td>
                <td>${e.dueDate ? format(parseISO(e.dueDate), 'MMM dd,yyyy') : 'N/A'}</td>
                <td>${e.status}</td>
                <td>${e.isCompleted && e.completionDate ? format(e.completionDate.toDate(), 'MMM dd,yyyy') : 'N/A'}</td>
                <td>${e.notes ? e.notes.length : 0}</td>
              </tr>
            `).join('')}
          </tbody>
        </table>`;
      } else {
        htmlContent += `<p>No experiments data available.</p>`;
      }

      // Staff Section
      htmlContent += `<h2>Staff Directory</h2>`;
      if (staffData.length > 0) {
        htmlContent += `<table>
          <thead>
            <tr>
              <th>Full Name</th>
              <th>Role</th>
              <th>Department</th>
              <th>Email</th>
              <th>Join Date</th>
            </tr>
          </thead>
          <tbody>
            ${staffData.map(s => `
              <tr>
                <td>${s.fullName}</td>
                <td>${s.role}</td>
                <td>${s.department}</td>
                <td>${s.email}</td>
                <td>${s.joinDate ? format(parseISO(s.joinDate), 'MMM dd,yyyy') : 'N/A'}</td>
              </tr>
            `).join('')}
          </tbody>
        </table>`;
      } else {
        htmlContent += `<p>No staff data available.</p>`;
      }

      htmlContent += `</body></html>`;

      const blob = new Blob([htmlContent], { type: 'application/msword' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `biotrack_lab_data_export_${format(new Date(), 'yyyyMMdd_HHmmss')}.doc`;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);

    } catch (err) {
      console.error("Error exporting to Word:", err);
      setExportError(`Failed to export Word document: ${err.message}`);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="p-6 bg-gray-800 rounded-lg shadow-lg">
      <h3 className="text-xl font-semibold mb-4 text-white">Data Export</h3>
      <p className="mb-4 text-gray-300">Download your lab data for Samples, Inventory, Experiments, and Staff.</p>
      <div className="flex flex-col sm:flex-row space-y-3 sm:space-y-0 sm:space-x-4">
        <button
          onClick={exportToPDF}
          className="bg-red-600 hover:bg-red-700 text-white font-bold py-2 px-4 rounded-md flex-1 transition duration-200 flex items-center justify-center space-x-2"
          disabled={loading}
        >
          {loading ? (
            <>
              <svg className="animate-spin h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
                <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
                <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
              </svg>
              <span>Generating PDF...</span>
            </>
          ) : (
            <>
              <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" viewBox="0 0 20 20" fill="currentColor">
                <path fillRule="evenodd" d="M4 4a2 2 0 012-2h4.586A2 2 0 0112 2.586L15.414 6A2 2 0 0116 7.414V16a2 2 0 01-2 2H6a2 2 0 01-2-2V4zm2 6a1 1 0 011-1h6a1 1 0 110 2H7a1 1 0 01-1-1zm1 3a1 1 0 100 2h6a1 1 0 100-2H7z" clipRule="evenodd" />
              </svg>
              <span>Download PDF</span>
            </>
          )}
        </button>
        <button
          onClick={exportToWord}
          className="bg-blue-600 hover:bg-blue-700 text-white font-bold py-2 px-4 rounded-md flex-1 transition duration-200 flex items-center justify-center space-x-2"
          disabled={loading}
        >
          {loading ? (
            <>
              <svg className="animate-spin h-5 w-5 text-white" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
                <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4"></circle>
                <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
              </svg>
              <span>Generating Word...</span>
            </>
          ) : (
            <>
              <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" viewBox="0 0 20 20" fill="currentColor">
                <path d="M9 2a1 1 0 000 2h2a1 1 0 100-2H9z" />
                <path fillRule="evenodd" d="M4 5a2 2 0 012-2 3 3 0 003 3h2a3 3 0 003-3 2 2 0 012 2v11a2 2 0 01-2 2H6a2 2 0 01-2-2V5zm3 4a1 1 0 000 2h.01a1 1 0 100-2H7zm3 0a1 1 0 000 2h.01a1 1 0 100-2H10zm3 0a1 1 0 000 2h.01a1 1 0 100-2H13zm-3 4a1 1 0 100 2h.01a1 1 0 100-2H10zm3 0a1 1 0 100 2h.01a1 1 0 100-2H13z" clipRule="evenodd" />
              </svg>
              <span>Download Word</span>
            </>
          )}
        </button>
      </div>
      {exportError && <p className="text-red-400 text-sm italic mt-4 text-center">{exportError}</p>}
    </div>
  );
};


// --- Main App Component ---
const App = () => {
  const { auth, userId, loading, authError } = useFirebase();
  const [currentPage, setCurrentPage] = useState('dashboard'); // Default page to dashboard

  if (loading) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-gray-900">
        <p className="text-lg text-gray-300">Initializing BioTrack Lab System...</p>
      </div>
    );
  }

  if (authError) {
    return (
      <div className="min-h-screen flex items-center justify-center bg-gray-900">
        <div className="text-center p-8 bg-gray-800 rounded-lg shadow-lg border border-gray-700">
          <p className="text-red-400 font-bold mb-4">Error: {authError}</p>
          <p className="text-gray-300">Please ensure Firebase is correctly configured.</p>
        </div>
      </div>
    );
  }

  if (!userId || !auth.currentUser) {
    return <Auth onLoginSuccess={() => setCurrentPage('dashboard')} />;
  }

  const handleLogout = async () => {
    try {
      await signOut(auth);
      console.log("User logged out.");
    } catch (error) {
      console.error("Error logging out:", error);
    }
  };

  const renderPage = () => {
    switch (currentPage) {
      case 'dashboard':
        return <Dashboard />;
      case 'samples':
        return <SampleTracking />;
      case 'inventory':
        return <InventoryManagement />;
      case 'experiments':
        return <Experiments />;
      case 'staff':
        return <StaffManagement />;
      case 'export':
        return <ExportData />;
      default:
        return <Dashboard />;
    }
  };

  return (
    <div className="min-h-screen bg-gray-900 font-sans text-gray-100">
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Inter:wght@400;500;600;700;800;900&display=swap');
        body {
          font-family: 'Inter', sans-serif;
        }
      `}</style>
      {/* CDN for html2canvas and jspdf */}
      <script src="https://cdnjs.cloudflare.com/ajax/libs/html2canvas/1.4.1/html2canvas.min.js"></script>
      <script src="https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js"></script>

      <div className="bg-gray-800 text-white p-4 shadow-xl flex flex-col sm:flex-row justify-between items-center rounded-b-lg border-b border-gray-700">
        <h1 className="text-3xl font-extrabold mb-2 sm:mb-0">BioTrack Lab System Management</h1>
        <div className="flex items-center space-x-4">
          <p className="text-sm text-gray-300">User ID: <span className="font-mono text-blue-300">{userId.substring(0, 8)}...</span></p>
          <button
            onClick={handleLogout}
            className="bg-red-600 hover:bg-red-700 text-white px-4 py-2 rounded-md text-sm font-semibold shadow-md transition duration-200"
          >
            Logout
          </button>
        </div>
      </div>

      <nav className="bg-gray-800 p-3 shadow-md mx-4 mt-4 rounded-lg border border-gray-700">
        <ul className="flex flex-wrap justify-center space-x-2 sm:space-x-4">
          <li>
            <button
              onClick={() => setCurrentPage('dashboard')}
              className={`px-4 py-2 rounded-md text-sm font-medium transition duration-200 ${
                currentPage === 'dashboard' ? 'bg-blue-600 text-white shadow-sm' : 'text-gray-300 hover:bg-gray-700'
              }`}
            >
              Dashboard
            </button>
          </li>
          <li>
            <button
              onClick={() => setCurrentPage('samples')}
              className={`px-4 py-2 rounded-md text-sm font-medium transition duration-200 ${
                currentPage === 'samples' ? 'bg-blue-600 text-white shadow-sm' : 'text-gray-300 hover:bg-gray-700'
              }`}
            >
              Sample Tracking
            </button>
          </li>
          <li>
            <button
              onClick={() => setCurrentPage('inventory')}
              className={`px-4 py-2 rounded-md text-sm font-medium transition duration-200 ${
                currentPage === 'inventory' ? 'bg-blue-600 text-white shadow-sm' : 'text-gray-300 hover:bg-gray-700'
              }`}
            >
              Inventory Management
            </button>
          </li>
          <li>
            <button
              onClick={() => setCurrentPage('experiments')}
              className={`px-4 py-2 rounded-md text-sm font-medium transition duration-200 ${
                currentPage === 'experiments' ? 'bg-blue-600 text-white shadow-sm' : 'text-gray-300 hover:bg-gray-700'
              }`}
            >
              Experiments
            </button>
          </li>
          <li>
            <button
              onClick={() => setCurrentPage('staff')}
              className={`px-4 py-2 rounded-md text-sm font-medium transition duration-200 ${
                currentPage === 'staff' ? 'bg-blue-600 text-white shadow-sm' : 'text-gray-300 hover:bg-gray-700'
              }`}
            >
              Staff
            </button>
          </li>
          <li>
            <button
              onClick={() => setCurrentPage('export')}
              className={`px-4 py-2 rounded-md text-sm font-medium transition duration-200 ${
                currentPage === 'export' ? 'bg-blue-600 text-white shadow-sm' : 'text-gray-300 hover:bg-gray-700'
              }`}
            >
              Export Data
            </button>
          </li>
        </ul>
      </nav>

      <main className="p-4">
        <div className="max-w-6xl mx-auto">
          {renderPage()}
        </div>
      </main>
    </div>
  );
};

// Wrap the main App component with the FirebaseProvider
export default function ProvidedApp() {
  return (
    <FirebaseProvider>

  );
